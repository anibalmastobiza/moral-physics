// --- V2: PHYSICS ENGINE WITH "MORAL TETHER" --- var physics_trial = { type: jsPsychCallFunction, async: true, func: function(done) { // 1. Setup Visuals const display = document.getElementById('jspsych-target'); display.innerHTML = `
Drag the Weight
`; const canvas = document.getElementById('moralCanvas'); const ctx = canvas.getContext('2d'); // --- EXPERIMENTAL CONFIG --- // "High Friction" = Cuerda floja y objeto pesado (difícil de arrastrar) // "Low Friction" = Cuerda tensa y objeto ligero (respuesta inmediata) const condition = Math.random() > 0.5 ? 'high_friction' : 'low_friction'; // Physics Constants (Spring System) const springStiffness = condition === 'high_friction' ? 0.03 : 0.15; const friction = 0.85; // Air resistance (prevents oscillation forever) // State Variables let startX = 400, startY = 500; let mouse = { x: startX, y: startY }; // Intention let agent = { x: startX, y: startY, vx: 0, vy: 0 }; // Action (pos, velocity) let trajectory = []; let isInteract = false; let trialRunning = true; // Flag to prevent double submission let startTime = performance.now(); // Targets const targetSelf = { x: 150, y: 150, r: 70, label: "KEEP ($10)", color: "#ef4444" }; const targetShare = { x: 650, y: 150, r: 70, label: "SHARE ($5/$5)", color: "#22c55e" }; // Input Handler function updateMouse(e) { const rect = canvas.getBoundingClientRect(); mouse.x = e.clientX - rect.left; mouse.y = e.clientY - rect.top; isInteract = true; } // We use document listener to catch movements even if mouse leaves canvas momentarily document.addEventListener('mousemove', updateMouse); // --- THE PHYSICS LOOP --- function loop() { if (!trialRunning) return; // 1. Calculate Forces (Hooke's Law: F = k * x) // Distance between intention (mouse) and action (agent) let dx = mouse.x - agent.x; let dy = mouse.y - agent.y; // Acceleration = Force (mass is assumed 1) let ax = dx * springStiffness; let ay = dy * springStiffness; // 2. Update Velocity with Friction/Drag agent.vx = (agent.vx + ax) * friction; agent.vy = (agent.vy + ay) * friction; // 3. Update Position agent.x += agent.vx; agent.y += agent.vy; // 4. Data Recording (60Hz) if (isInteract) { trajectory.push({ t: Math.round(performance.now() - startTime), ax: Math.round(agent.x), ay: Math.round(agent.y), // Actual mx: Math.round(mouse.x), my: Math.round(mouse.y) // Intention }); } // 5. Render render(); // 6. Logic (Collision) checkWin(); requestAnimationFrame(loop); } function render() { ctx.clearRect(0, 0, canvas.width, canvas.height); // Draw Targets drawTarget(targetSelf); drawTarget(targetShare); // Draw The Tether (The "Moral Tension") ctx.beginPath(); ctx.moveTo(agent.x, agent.y); ctx.lineTo(mouse.x, mouse.y); // Color changes based on tension (strain) let dist = Math.hypot(mouse.x - agent.x, mouse.y - agent.y); let tensionColor = dist > 200 ? "rgba(255,0,0,0.6)" : "rgba(0,0,0,0.2)"; ctx.strokeStyle = tensionColor; ctx.lineWidth = 3; ctx.setLineDash([5, 5]); // Dashed line implies elasticity ctx.stroke(); ctx.setLineDash([]); // Draw Agent (The "Heavy" Body) ctx.beginPath(); ctx.arc(agent.x, agent.y, 20, 0, Math.PI * 2); ctx.fillStyle = "#2563eb"; // Blue ctx.fill(); ctx.fillStyle = "white"; ctx.font = "10px Arial"; ctx.textAlign = "center"; ctx.fillText("ACT", agent.x, agent.y+4); // Draw "Ghost" Cursor (Intention) - Crucial fix: user sees their hand ctx.beginPath(); ctx.arc(mouse.x, mouse.y, 8, 0, Math.PI * 2); ctx.strokeStyle = "#ef4444"; // Red outline ctx.lineWidth = 2; ctx.stroke(); } function drawTarget(t) { ctx.beginPath(); ctx.arc(t.x, t.y, t.r, 0, Math.PI * 2); ctx.fillStyle = t.color + "22"; // Very transparent ctx.fill(); ctx.strokeStyle = t.color; ctx.lineWidth = 2; ctx.stroke(); ctx.fillStyle = "#333"; ctx.font = "bold 16px sans-serif"; ctx.fillText(t.label, t.x, t.y); } function checkWin() { // Check Selfish if (dist(agent, targetSelf) < 30) { finishTrial('selfish'); } // Check Altruistic else if (dist(agent, targetShare) < 30) { finishTrial('altruistic'); } } function dist(p1, p2) { return Math.hypot(p1.x - p2.x, p1.y - p2.y); } function finishTrial(outcome) { if (!trialRunning) return; trialRunning = false; // Stop the loop immediately document.removeEventListener('mousemove', updateMouse); // Visual Feedback on success const canvas = document.getElementById('moralCanvas'); const ctx = canvas.getContext('2d'); ctx.fillStyle = "rgba(255, 255, 255, 0.8)"; ctx.fillRect(0, 0, 800, 600); ctx.fillStyle = "black"; ctx.font = "30px sans-serif"; ctx.fillText("Choice Recorded: " + outcome.toUpperCase(), 400, 300); // Save Data var trial_data = { condition: condition, outcome: outcome, rt: performance.now() - startTime, // Tension Metric: Average distance between mouse and agent avg_tension: trajectory.reduce((acc, p) => acc + Math.hypot(p.mx-p.ax, p.my-p.ay), 0) / trajectory.length, trajectory: JSON.stringify(trajectory) // Ensure it's stringified for Sheets }; // Wait 1 second then advance setTimeout(() => { jsPsych.finishTrial(trial_data); }, 1000); } // Start loop(); } };
